```python
# Bootstrap CI for correlation
from scipy import stats

def bootstrap_correlation_ci(x, y, n_bootstrap=1000, ci=0.95):
    correlations = []
    n = len(x)
    for _ in range(n_bootstrap):
        indices = np.random.choice(n, n, replace=True)
        r, _ = stats.pearsonr(x[indices], y[indices])
        correlations.append(r)
    lower = np.percentile(correlations, (1-ci)/2 * 100)
    upper = np.percentile(correlations, (1+ci)/2 * 100)
    return lower, upper

ci_low, ci_high = bootstrap_correlation_ci(
    submissions['Survivors'].values,
    submissions['Score'].values
)
print(f"Correlation: r = {corr:.2f}, 95% CI: [{ci_low:.2f}, {ci_high:.2f}]")
```
